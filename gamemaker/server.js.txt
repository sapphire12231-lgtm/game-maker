// Simple per-visitor limit: 10 games per hour
const LIMIT = 10, WINDOW = 3600e3;
const hits = new Map();
setInterval(function () {
  const now = Date.now();
  for (const [ip, list] of hits) {
    const fresh = list.filter(function (t) { return now - t < WINDOW; });
    if (fresh.length) hits.set(ip, fresh); else hits.delete(ip);
  }
}, 600e3).unref();

function limited(ip) {
  const now = Date.now();
  const list = (hits.get(ip) || []).filter(function (t) { return now - t < WINDOW; });
  if (list.length >= LIMIT) { hits.set(ip, list); return true; }
  list.push(now); hits.set(ip, list); return false;
}

function send(res, code, obj) {
  res.writeHead(code, { 'content-type': 'application/json' });
  res.end(JSON.stringify(obj));
}

function clientIp(req) {
  const xff = String(req.headers['x-forwarded-for'] || '').split(',').map(function (s) { return s.trim(); }).filter(Boolean);
  return xff.length ? xff[xff.length - 1] : (req.socket.remoteAddress || 'unknown');
}

const server = http.createServer(async function (req, res) {
  try {
    const path = req.url.split('?')[0];

    if (req.method === 'GET' && (path === '/' || path === '/index.html')) {
      res.writeHead(200, { 'content-type': 'text/html; charset=utf-8' });
      return res.end(PAGE);
    }

    if (req.method === 'POST' && path === '/api/generate') {
      if (limited(clientIp(req))) return send(res, 429, { error: 'Too many games. Try again in an hour.' });

      let body = '';
      for await (const chunk of req) {
        body += chunk;
        if (body.length > 2000) return send(res, 413, { error: 'Request too long.' });
      }
      let idea = '';
      try { idea = String(JSON.parse(body).idea || '').trim().slice(0, 300); }
      catch (e) { return send(res, 400, { error: 'Bad request.' }); }
      if (!idea) return send(res, 400, { error: 'Type a game idea.' });

      const r = await fetch('https://api.anthropic.com/v1/messages', {
        method: 'POST',
        headers: { 'content-type': 'application/json', 'x-api-key': KEY, 'anthropic-version': '2023-06-01' },
        body: JSON.stringify({
          model: 'claude-sonnet-5-5',
          max_tokens: 12000,
          system: SYS,
          messages: [{ role: 'user', content: 'Game: ' + idea }]
        })
      });
      const d = await r.json();
      if (!r.ok) { console.error('API error:', JSON.stringify(d)); return send(res, 502, { error: 'The AI had a problem. Try again.' }); }
      const html = d.content.map(function (c) { return c.text || ''; }).join('')
        .replace(/^```(?:html)?\s*/i, '').replace(/```\s*$/, '').trim();
      return send(res, 200, { html: html, truncated: d.stop_reason === 'max_tokens' });
    }

    res.writeHead(404, { 'content-type': 'text/plain' });
    res.end('Not found');
  } catch (e) {
    console.error(e);
    if (!res.headersSent) send(res, 500, { error: 'Server error. Try again.' });
  }
});

server.requestTimeout = 0;
server.listen(process.env.PORT || 3000, function () {
  console.log('Running on port ' + (process.env.PORT || 3000));
});
